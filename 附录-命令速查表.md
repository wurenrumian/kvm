# 附录 · 命令速查表

> 一页纸备查。所有命令在 Fedora 44 + libvirt 环境下验证过语义。

## 一、环境与检查

```bash
# 硬件虚拟化能力
grep -m1 -oE 'vmx|svm' /proc/cpuinfo
lsmod | grep -E '^kvm'
ls -l /dev/kvm
systemd-detect-virt

# 安装
sudo dnf install -y qemu-kvm libvirt-daemon-kvm virt-install libguestfs-tools
sudo systemctl enable --now libvirtd

# 版本
qemu-system-x86_64 --version
virsh version
```

## 二、QEMU 裸命令

```bash
# 最简启动
qemu-system-x86_64 -enable-kvm -cpu host -smp 2 -m 2048 \
  -drive file=disk.qcow2,if=virtio,format=qcow2 -nographic

# 挂 ISO 安装
qemu-system-x86_64 -enable-kvm -cpu host -smp 2 -m 2048 \
  -cdrom install.iso -boot d -nographic

# 常用参数
#   -enable-kvm        KVM 加速
#   -cpu host          CPU 透传
#   -smp N             vCPU 数
#   -m MB              内存
#   -drive file=...,if=virtio,format=qcow2
#   -netdev user,id=n0 -device virtio-net-pci,netdev=n0
#   -nographic         串口直连
#   -daemonize         后台运行
#   -monitor stdio     QEMU monitor
#   Ctrl-a x           退出
```

## 四、virt-install

```bash
sudo virt-install \
  --name vm1 --memory 2048 --vcpus 2 \
  --cpu host-passthrough \
  --disk path=/var/lib/libvirt/images/vm1.qcow2,size=10,bus=virtio \
  --cdrom /path/install.iso \
  --network network=default,model=virtio \
  --graphics none --console pty,target_type=serial

# 从 cloud 镜像导入
sudo virt-install --name c1 --memory 2048 --vcpus 2 --import \
  --disk path=/var/lib/libvirt/images/c1.qcow2,bus=virtio \
  --cloud-init root-password-generate=on \
  --network network=default,model=virtio \
  --graphics none --console pty,target_type=serial
```

## 十二、性能调优速查

```bash
# CPU
virsh vcpupin vm1 0 2
virsh emulatorpin vm1 0-1
numactl --hardware
taskset -pc <pid>

# 内存
grep HugePages /proc/meminfo
echo 4 | sudo tee /sys/kernel/mm/hugepages/hugepages-1048576kB/nr_hugepages
echo 0 | sudo tee /sys/kernel/mm/ksm/run

# 磁盘/网络压测
fio --name=t --ioengine=libaio --rw=randread --bs=4k --iodepth=32 --numjobs=4 --size=1G --runtime=30 --group_reporting
iperf3 -c <ip> -P 8
stress-ng --cpu 4 --timeout 60s
```

## 十三、直通与安全

```bash
# IOMMU / VFIO
dmesg | grep -i -e DMAR -e IOMMU
for d in /sys/kernel/iommu_groups/*/devices/*; do \
  echo "Group $(basename $(dirname $(dirname $d))): $(basename $d)"; done

# sVirt / SELinux
getenforce
ps -eZ | grep qemu
ls -Z /var/lib/libvirt/images/*.qcow2
sudo semanage fcontext -a -t svirt_image_t "/data/vms(/.*)?"
sudo restorecon -Rv /data/vms

# 嵌套虚拟化
cat /sys/module/kvm_intel/parameters/nested
sudo modprobe kvm_intel nested=1
```

## 十四、常见 XML 片段

```xml
<!-- 磁盘：高性能 -->
<disk type='file' device='disk'>
  <driver name='qemu' type='qcow2' cache='none' io='io_uring' discard='unmap' queues='4'/>
  <source file='/var/lib/libvirt/images/vm1.qcow2'/>
  <target dev='vda' bus='virtio'/>
</disk>

<!-- 网卡：多队列 + vhost -->
<interface type='network'>
  <source network='default'/>
  <model type='virtio'/>
  <driver name='vhost' queues='4'/>
</interface>

<!-- CPU：绑核 -->
<vcpu placement='static'>4</vcpu>
<cputune>
  <vcpupin vcpu='0' cpuset='2'/>
  <vcpupin vcpu='1' cpuset='3'/>
  <emulatorpin cpuset='0-1'/>
</cputune>

<!-- 内存：巨页 -->
<memoryBacking><hugepages/></memoryBacking>

<!-- 直通 -->
<hostdev mode='subsystem' type='pci' managed='yes'>
  <source><address domain='0x0000' bus='0x01' slot='0x00' function='0x0'/></source>
</hostdev>

<!-- vTPM -->
<tpm model='tpm-crb'><backend type='emulator' version='2.0'/></tpm>
```
